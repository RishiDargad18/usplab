#include<stdio.h>
#include<stdlib.h>
#include<unistd.h>
#include<sys/wait.h>
int main(void) {
    pid_t pid, pid2, pid3;
    if ((pid = fork()) < 0) {
        printf("fork error\n");
    }
    else if (pid == 0) {   // first child
        if ((pid3 = fork()) < 0) {
            printf("fork error\n");
        }
        else if (pid3 == 0) {   // second child
            sleep(5);
            printf("Second child PID: %d\n", getpid());
            printf("Second child parent PID: %ld\n", (long)getppid());
            exit(0);
        }
        // First child exits immediately → creates zombie (for second child temporarily)
        printf("First child PID: %d\n", getpid());
        exit(0);
    }
    // Parent waits → removes zombie of first child
    pid2 = waitpid(pid, NULL, 0);
    if (pid2 != pid)
        printf("waitpid error\n");
    printf("Terminated child PID: %d\n", pid2);
    return 0;
}
