#include <stdio.h>
#include <unistd.h>
#include <sys/wait.h>
#include <stdlib.h>
int main() {
    pid_t pid = fork();
    if (pid == 0) {
        printf("Child PID: %d\n", getpid());
        exit(0); 
    }
    sleep(5);   
    wait(NULL);
    printf("Zombie removed\n");
    return 0;
}
