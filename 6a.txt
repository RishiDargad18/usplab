#include <stdio.h>
#include <unistd.h>
#include <fcntl.h>
int main() {
    int fd1 = open("example.txt", O_CREAT | O_RDWR, 0666);
    int fd2 = open("sample.txt", O_CREAT | O_RDWR, 0666);
    fd2 = dup2(fd1, fd2);
    char buf[20];
    read(fd1, buf, 20);
    write(fd2, buf, 20);
    printf("File descriptors: %d %d\n", fd1, fd2);
    printf("%.*s\n", 20, buf);
    close(fd1);
    close(fd2);
    return 0;
}
