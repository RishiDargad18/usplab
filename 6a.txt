#include <stdio.h>
#include <unistd.h>
#include <fcntl.h>
#include <stdlib.h>
int main(int argc, char *argv[]) {
    int n = atoi(argv[2]);
    int fd1 = open(argv[1], O_RDWR);
    int fd2 = dup2(fd1, 5);
    char buf[100];
    int bytes = read(fd1, buf, n);
    lseek(fd2, 0, SEEK_END);
    write(fd2, buf, bytes);
    close(fd1);
    close(fd2);
    return 0;
}
