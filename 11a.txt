Dup1.c
#include <stdio.h>
#include <unistd.h>
#include <fcntl.h>
int main() {
    int fd1 = open("t12.txt", O_RDWR | O_CREAT, 0666);
    char buf[] = "abcdef";
    int fd2 = dup(fd1);
    printf("%d %d\n", fd1, fd2);
    write(fd1, buf, 6);
    return 0;
}

Dup2.c
#include <stdio.h>
#include <unistd.h>
#include <fcntl.h>

int main() {
    int fd11 = open("t12.txt", O_RDWR | O_CREAT, 0666);
    char buf[] = "abcdef";
    int fd12 = 0;   // stdin
    dup2(fd11, fd12);
    printf("%d %d\n", fd11, fd12);
    write(fd12, buf, 6);
    return 0;
}
