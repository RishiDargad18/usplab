#include <stdio.h>
#include <unistd.h>
#include <sys/wait.h>
int main() {
    pid_t pid;
    char *env[] = {"USER=unknown", "PATH=/tmp", NULL};
    if ((pid = fork()) == 0)
        execle("./echoall", "echoall", "child", (char *)0, env);
    wait(NULL);
    execlp("./echoall", "echoall", "parent", (char *)0);
}
