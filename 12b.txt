#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/wait.h>
void err_sys(const char *msg){
    perror(msg);
    exit(1);
}
int main(){
    pid_t pid;
    char *env_init[] = {
        "USER=unknown",
        "PATH=/tmp",
        NULL
    };
    if ((pid = fork()) < 0){
        err_sys("fork");
    }
    else if (pid == 0){
        // Child → custom environment
        execle("./echoall", "echoall", "myarg1", "MY ARG2", (char *)0, env_init);
        err_sys("execle");
    }
    // Parent waits
    if (waitpid(pid, NULL, 0) < 0)
        err_sys("waitpid");
    // Parent → default environment
    execlp("./echoall", "echoall", "only 1 arg", (char *)0);
    err_sys("execlp");
    return 0;
}

#include <stdio.h>
extern char **environ;
int main(int argc, char *argv[]){
    for (int i = 0; i < argc; i++)
        printf("argv[%d]= %s\n", i, argv[i]);
    printf("\nEnvironment variables:\n");
    for (char **env = environ; *env != NULL; env++)
        printf("%s\n", *env);
    return 0;
}
