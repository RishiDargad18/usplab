#include <stdio.h>
#include <sys/stat.h>
#include <time.h>
int main(int argc, char *argv[]) {
    struct stat s;
    if (stat(argv[1], &s) < 0) return 1;
    printf("File: %s\n", argv[1]);
    printf("Size: %lld bytes\n", (long long)s.st_size);
    printf("Permissions: %o\n", s.st_mode & 0777);
    printf("Number of Links: %ld\n", (long)s.st_nlink);
    printf("Owner: UID=%ld, GID=%ld\n", (long)s.st_uid, (long)s.st_gid);
    printf("Last Access Time: %s", ctime(&s.st_atime));
    return 0;
}
