#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <fcntl.h>
#include <string.h>
int main(int argc, char *argv[]) {
    int fd = open(argv[1], O_RDWR);
    if (fd < 0) {
        perror("open");
        return 1;
    }
    struct flock lock;
    off_t size = lseek(fd, 0, SEEK_END);
    lock.l_whence = SEEK_SET;
    lock.l_start = (size < 100) ? 0 : size - 100;
    lock.l_len  = (size < 100) ? size : 100;
    lock.l_type = F_WRLCK;
    fcntl(fd, F_GETLK, &lock);
    if (lock.l_type != F_UNLCK) {
        printf("Locked by PID: %d\n", lock.l_pid);
    } else {
        printf("Not locked. Locking...\n");
        lock.l_type = F_WRLCK;
        fcntl(fd, F_SETLK, &lock);
        char buf[51] = {0};
        lseek(fd, (size < 50) ? 0 : size - 50, SEEK_SET);
        read(fd, buf, (size < 50) ? size : 50);
        printf("Last 50 bytes:\n%s\n", buf);
        sleep(10);
        lock.l_type = F_UNLCK;
        fcntl(fd, F_SETLK, &lock);
        printf("Unlocked\n");
    }
    close(fd);
    return 0;
}
