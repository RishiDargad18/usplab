#include <stdio.h>
#include <fcntl.h>
#include <unistd.h>
int main(int argc, char *argv[]) {
    if (argc < 2) {
        fprintf(stderr, "Usage: %s <filename>\n", argv[0]);
        return 1;
    }
    int fd = open(argv[1], O_RDONLY);
    if (fd < 0) {
        perror("open");
        return 1;
    }
    char buf[21];
    // i) First 20 chars
    if (read(fd, buf, 20) != 20) { perror("read"); close(fd); return 1; }
    buf[20] = '\0';
    printf("First 20 characters: %s\n", buf);
    // ii) From 10th byte
    if (lseek(fd, 10, SEEK_SET) < 0) { perror("lseek"); close(fd); return 1; }
    if (read(fd, buf, 20) != 20) { perror("read"); close(fd); return 1; }
    buf[20] = '\0';
    printf("20 characters from 10th byte: %s\n", buf);
    // iii) 10 bytes ahead from current
    if (lseek(fd, 10, SEEK_CUR) < 0) { perror("lseek"); close(fd); return 1; }
    if (read(fd, buf, 20) != 20) { perror("read"); close(fd); return 1; }
    buf[20] = '\0';
    printf("20 characters from current offset: %s\n", buf);
    // iv) File size
    off_t sz = lseek(fd, 0, SEEK_END);
    if (sz < 0) { perror("lseek"); close(fd); return 1; }
    printf("File size: %lld bytes\n", (long long)sz);
    close(fd);
    return 0;
}
