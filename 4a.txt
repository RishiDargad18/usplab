#include <stdio.h>
#include <unistd.h>
int main(int argc, char *argv[]){
    if (argc == 3) {
        if (!link(argv[1], argv[2]))
            printf("Hard link created\n");
        else
            printf("Hard link error\n");
    }
    else if (argc == 4) {
        if (!symlink(argv[2], argv[3]))
            printf("Soft link created\n");
    }
    return 0;
}
