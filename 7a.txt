#include <stdio.h>
#include <setjmp.h>
#include <stdlib.h>
static jmp_buf jb;
static int gv;
void f2(){ longjmp(jb, 1); }
void f1(int a, int r, int v, int s){
    printf("in f1()\n");
    printf("gv=%d,av=%d,rv=%d,vv=%d,sv=%d\n", gv, a, r, v, s);
    f2();
}
int main(){
    int av = 2;
    register int rv = 3;
    volatile int vv = 4;
    static int sv = 5;
    gv = 1;
    if (setjmp(jb)){
        printf("after longjmp\n");
        printf("gv=%d,av=%d,rv=%d,vv=%d,sv=%d\n", gv, av, rv, vv, sv);
        return 0;
    }
    gv = 95; av = 96; rv = 97; vv = 98; sv = 99;
    f1(av, rv, vv, sv);
}
