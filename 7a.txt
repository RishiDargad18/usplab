#include <stdio.h>
#include <setjmp.h>
static jmp_buf jb;
static int gv;
int main() {
    int av = 2;
    register int rv = 3;
    volatile int vv = 4;
    static int sv = 5;
    gv = 1;
    if (setjmp(jb)) {
        printf("after longjmp\n");
        printf("gv=%d av=%d rv=%d vv=%d sv=%d\n", gv, av, rv, vv, sv);
        return 0;
    }
    gv = 95; av = 96; rv = 97; vv = 98; sv = 99;
    printf("before longjmp\n");
    printf("gv=%d av=%d rv=%d vv=%d sv=%d\n", gv, av, rv, vv, sv);
    longjmp(jb, 1);
}
