#include <stdio.h>
#include <fcntl.h>
#include <unistd.h>
#include <stdlib.h>
#define BUF 1024
int main(int argc, char *argv[]) {
    if (argc < 3) {
        fprintf(stderr, "Usage: %s <source> <destination>\n", argv[0]);
        return 1;
    }
    int s = open(argv[1], O_RDONLY);
    if (s < 0) { perror("open source"); return 1; }
    int d = open(argv[2], O_WRONLY | O_CREAT | O_TRUNC, 0600);
    if (d < 0) { perror("open destination"); close(s); return 1; }
    char buf[BUF];
    ssize_t n;
    while ((n = read(s, buf, BUF)) > 0)
        if (write(d, buf, n) != n) {
            perror("write");
            goto end;
        }
    if (n < 0) perror("read");
end:
    if (close(s) < 0) perror("close source");
    if (close(d) < 0) perror("close destination");
    return 0;
}
