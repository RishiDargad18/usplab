#include <stdio.h>
#include <fcntl.h>
#include <unistd.h>
#include <stdlib.h>
int main(int argc, char *argv[]) {
    int s = open(argv[1], O_RDONLY);
    if (s < 0) {
        perror("open source");
        return 1;
    }
    int d = open(argv[2], O_WRONLY | O_CREAT | O_TRUNC, 0600);
    if (d < 0) {
        perror("open destination");
        close(s);
        return 1;
    }
    char buf[1024];
    ssize_t n;
    while (1) {
        n = read(s, buf, 1024);
        if (n > 0) {
            ssize_t w = write(d, buf, n);
            if (w != n) {
                perror("write");
                goto end;
            }
        }
        else if (n == 0) {
            break;
        }
        else {
            perror("read");
            goto end;
        }
    }
end:
    if (close(s) < 0) {
        perror("close source");
    }
    if (close(d) < 0) {
        perror("close destination");
    }
    return 0;
}
