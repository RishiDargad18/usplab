#include <stdio.h>
#include <fcntl.h>
#include <unistd.h>
int main(int argc, char *argv[]) {
    int fd = open(argv[1], O_RDONLY);
    if (fd < 0) return 1;
    int size = lseek(fd, 0, SEEK_END);
    if (size < 0) return 1;
    char c;
    for (int i = 1; i <= size; i++) {
        lseek(fd, -i, SEEK_END);
        read(fd, &c, 1);
        putchar(c);
    }
    printf("\n");
    close(fd);
    return 0;
}
